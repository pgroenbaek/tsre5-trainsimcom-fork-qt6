cmake_minimum_required(VERSION 3.21)

project(TSRE5)

# Find includes in the build directories
set(CMAKE_INCLUDE_CURRENT_DIR ON)

# Turn on automatic invocation of the MOC, UIC & RCC
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)

# OpenGL preference
set(OpenGL_GL_PREFERENCE LEGACY)

# Find the libraries
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network OpenGL OpenGLWidgets WebSockets)
find_package(fmt CONFIG REQUIRED)
find_package(OpenAL CONFIG REQUIRED)
find_package(OpenGL REQUIRED)

# Automatically include all .cpp and .h files in the root folder
file(GLOB SRC_FILES
  "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/*.h"
  "${CMAKE_CURRENT_SOURCE_DIR}/*.qrc"
)

# Exclude any autogenerated moc_*.cpp files from sources list.
list(FILTER SRC_FILES EXCLUDE REGEX ".*moc_.*\\.cpp$")

# Add the .rc file on windows only.
if(WIN32)
    list(APPEND SRC_FILES TSRE5_resource.rc)
endif()

# Tell CMake to create the executable
add_executable(TSRE5 ${SRC_FILES})

# Set CXX properties
set_target_properties(TSRE5 PROPERTIES
    CXX_STANDARD 17
    CXX_STANDARD_REQUIRED YES
    CXX_EXTENSIONS NO
)

# Set target compile options
target_compile_options(TSRE5 PRIVATE
    -Wall
    -O2
)

# Add the required Qt6 modules and other libraries for linking
target_link_libraries(TSRE5
    PRIVATE
    Qt6::Core
    Qt6::Gui
    Qt6::Widgets
    Qt6::Network
    Qt6::OpenGL
    Qt6::OpenGLWidgets
    Qt6::WebSockets
    OpenAL::OpenAL
    fmt::fmt
    ${OPENGL_LIBRARIES}
)
