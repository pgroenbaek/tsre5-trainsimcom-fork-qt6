cmake_minimum_required(VERSION 3.21)

project(TSRE5)

# Find includes in the build directories
set(CMAKE_INCLUDE_CURRENT_DIR ON)

# Turn on automatic invocation of the MOC, UIC & RCC
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)

# Set OpenGL preference
set(OpenGL_GL_PREFERENCE LEGACY)

# Set CXX properties
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Find the libraries
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network OpenGL OpenGLWidgets WebSockets)
find_package(fmt REQUIRED)
find_package(OpenAL REQUIRED)
find_package(OpenGL REQUIRED)
find_package(OpenSSL REQUIRED)

# Automatically include all .cpp and .h files in the root folder
set_property(GLOBAL PROPERTY USE_FOLDERS ON)
file(GLOB_RECURSE SRC_FILES
  "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/*.h"
  "${CMAKE_CURRENT_SOURCE_DIR}/*.ui"
  "${CMAKE_CURRENT_SOURCE_DIR}/*.qrc"
)

# Exclude any autogenerated moc_*.cpp and CMake files from sources list.
list(FILTER SRC_FILES EXCLUDE REGEX ".*moc_.*\\.cpp$")
list(FILTER SRC_FILES EXCLUDE REGEX ".*/build/.*")
list(FILTER SRC_FILES EXCLUDE REGEX ".*/CMakeFiles/.*")
list(FILTER SRC_FILES EXCLUDE REGEX ".*CMakeCXXCompilerId\\.cpp$")

# Add the .rc file on windows only.
if(WIN32)
    list(APPEND SRC_FILES TSRE5_resource.rc)
endif()

# Tell CMake to create the executable
add_executable(TSRE5 ${SRC_FILES})

# Set compile options
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU")
    # GCC (Linux, MinGW)
    target_compile_options(TSRE5 PRIVATE
        -Wall
        -Wextra
        "$<$<CONFIG:Debug>:-g;-O0;-fno-omit-frame-pointer>" # Debug-only
        "$<$<CONFIG:Release>:-O2>" # Release-only
    )
    target_compile_definitions(TSRE5 PRIVATE
        "$<$<CONFIG:Debug>:_GLIBCXX_DEBUG>" # Debug-only
    )
elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    # Clang (Linux or macOS)
    target_compile_options(TSRE5 PRIVATE
        -Wall
        -Wextra
        "$<$<CONFIG:Debug>:-g;-O0;-fno-omit-frame-pointer>" # Debug-only
        "$<$<CONFIG:Release>:-O2>" # Release-only
    )
endif()

# Add the required Qt6 modules and other libraries for linking
target_link_libraries(TSRE5 PRIVATE
    Qt6::Core
    Qt6::Gui
    Qt6::Widgets
    Qt6::Network
    Qt6::OpenGL
    Qt6::OpenGLWidgets
    Qt6::WebSockets
    OpenAL::OpenAL
    fmt::fmt
    OpenSSL::SSL
    OpenSSL::Crypto
    ${OPENGL_LIBRARIES}
)

# Link extra system libraries needed on windows
if(WIN32)
    target_link_libraries(TSRE5 PRIVATE
        ws2_32      # Windows Sockets API for network communication
        iphlpapi    # IP Helper API for network configuration info
        dnsapi      # DNS API for domain name resolution
        crypt32     # Cryptography API for certificates, encryption, etc.
        ole32       # OLE Automation, COM support
        oleaut32    # OLE Automation (variant, BSTR, dispatch interfaces)
        winmm       # Windows Multimedia API (timers, audio, joystick)
        comdlg32    # Common dialogs (open/save file dialogs, print dialogs)
        imm32       # Input Method Manager (IME support)
        gdi32       # Graphics Device Interface (2D graphics, fonts)
        shell32     # Shell API functions (file operations, icons, etc.)
        user32      # Core Windows UI (windows, messages, keyboard/mouse)
        kernel32    # Core Windows OS functions (memory, processes, threads)
        mpr         # Multiple Provider Router (network connections)
        advapi32    # Advanced Windows API (registry, services, security)
        uuid        # Universally Unique Identifier (UUID) generation
    )
endif()

# Post-build commands for windows (copies everything needed into the /dist/<config> folder)
if(WIN32)
    set(POSTBUILD_ENV
        "QT_ROOT=$ENV{QT_ROOT}"
        "QT_VERSION=$ENV{QT_VERSION}"
        "QT_ARCH=$ENV{QT_ARCH}"
        "VCPKG_ROOT=$ENV{VCPKG_ROOT}"
        "VCPKG_TRIPLET=$ENV{VCPKG_TRIPLET}"
    )
    set(DEBUG_SCRIPT "$<SHELL_PATH:${CMAKE_SOURCE_DIR}/scripts/postbuild-debug-windows.bat>")
    set(RELEASE_SCRIPT "$<SHELL_PATH:${CMAKE_SOURCE_DIR}/scripts/postbuild-release-windows.bat>")

    add_custom_command(TARGET TSRE5
        POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E env
            ${POSTBUILD_ENV}
            "$<IF:$<CONFIG:Debug>,${DEBUG_SCRIPT},${RELEASE_SCRIPT}>"
        COMMENT "Running post-build steps for $<CONFIG>"
        VERBATIM
    )
endif()
